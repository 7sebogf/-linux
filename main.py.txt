from fastapi import FastAPI
from sqlalchemy import create_engine, text
from pydantic import BaseModel

app = FastAPI()
engine = create_engine("postgresql://postgres:mysecretpassword@localhost:5432/history_db")

class UserAuth(BaseModel):
    username: str
    password: str

@app.get("/articles")
def get_articles(topic: str = None, search: str = None):
    with engine.connect() as conn:
        # Базовый запрос
        base_query = """
            SELECT a.id, a.title, a.content, t.name as topic 
            FROM articles a 
            JOIN topics t ON a.topic_id = t.id
            WHERE 1=1
        """
        params = {}
        
        if topic:
            base_query += " AND t.name = :t"
            params["t"] = topic
        if search:
            # Ищем совпадение в заголовке или тексте (без учета регистра)
            base_query += " AND (a.title ILIKE :s OR a.content ILIKE :s)"
            params["s"] = f"%{search}%"
            
        res = conn.execute(text(base_query), params)
        return {"articles": [{"id": r[0], "title": r[1], "content": r[2], "topic": r[3]} for r in res]}

@app.post("/register")
def register(user: UserAuth):
    with engine.connect() as conn:
        try:
            conn.execute(text("INSERT INTO users (username, password) VALUES (:u, :p)"), 
                         {"u": user.username, "p": user.password})
            conn.commit()
            return {"status": "success", "message": "Пользователь создан!"}
        except Exception:
            return {"status": "error", "message": "Этот логин уже занят"}

@app.post("/login")
def login(user: UserAuth):
    with engine.connect() as conn:
        result = conn.execute(
            text("SELECT id FROM users WHERE username = :u AND password = :p"),
            {"u": user.username, "p": user.password}
        ).fetchone()
        
        if result:
            return {"status": "success", "message": "Вход выполнен!"}
        else:
            return {"status": "error", "message": "Неверный логин или пароль"}

class FavoriteRequest(BaseModel):
    username: str
    article_id: int

# Добавить в избранное
@app.post("/favorites/add")
def add_favorite(req: FavoriteRequest):
    with engine.connect() as conn:
        try:
            # Находим ID пользователя по его логину
            user_res = conn.execute(text("SELECT id FROM users WHERE username = :u"), {"u": req.username}).fetchone()
            if not user_res:
                return {"status": "error", "message": "Пользователь не найден"}
            
            user_id = user_res[0]
            
            # Добавляем связь в таблицу favorites
            conn.execute(text("INSERT INTO favorites (user_id, article_id) VALUES (:uid, :aid)"), 
                         {"uid": user_id, "aid": req.article_id})
            conn.commit()
            return {"status": "success", "message": "Добавлено в избранное"}
        except Exception:
            # Если уже добавлено, база выдаст ошибку (потому что у нас PRIMARY KEY user_id + article_id)
            return {"status": "error", "message": "Уже в избранном"}

# Получить список избранного для пользователя
@app.get("/favorites")
def get_favorites(username: str):
    with engine.connect() as conn:
        query = text("""
            SELECT a.id, a.title, a.content, t.name as topic 
            FROM articles a 
            JOIN topics t ON a.topic_id = t.id
            JOIN favorites f ON a.id = f.article_id
            JOIN users u ON f.user_id = u.id
            WHERE u.username = :u
        """)
        res = conn.execute(query, {"u": username})
        return {"articles": [{"id": r[0], "title": r[1], "content": r[2], "topic": r[3]} for r in res]}

class InterestRequest(BaseModel):
    username: str
    topic_name: str

# Сохранить интерес пользователя
@app.post("/interests/add")
def add_interest(req: InterestRequest):
    with engine.connect() as conn:
        try:
            # Ищем ID пользователя и ID темы по их названиям
            user_res = conn.execute(text("SELECT id FROM users WHERE username = :u"), {"u": req.username}).fetchone()
            topic_res = conn.execute(text("SELECT id FROM topics WHERE name = :t"), {"t": req.topic_name}).fetchone()
            
            if user_res and topic_res:
                user_id = user_res[0]
                topic_id = topic_res[0]
                # Записываем связь в таблицу
                conn.execute(text("INSERT INTO user_interests (user_id, topic_id) VALUES (:uid, :tid)"), 
                             {"uid": user_id, "tid": topic_id})
                conn.commit()
                return {"status": "success", "message": f"Интерес '{req.topic_name}' сохранен!"}
            return {"status": "error", "message": "Пользователь или тема не найдены"}
        except Exception:
            # Если такая связь уже есть, база выдаст ошибку (PRIMARY KEY)
            return {"status": "error", "message": "Этот интерес уже сохранен"}

# Получить интересы пользователя
@app.get("/interests")
def get_interests(username: str):
    with engine.connect() as conn:
        # Достаем названия тем, которые выбрал пользователь
        query = text("""
            SELECT t.name FROM topics t
            JOIN user_interests ui ON t.id = ui.topic_id
            JOIN users u ON ui.user_id = u.id
            WHERE u.username = :u
        """)
        res = conn.execute(query, {"u": username})
        return {"interests": [row[0] for row in res]}

@app.delete("/favorites/remove")
def remove_favorite(username: str, article_id: int):
    with engine.connect() as conn:
        try:
            user_res = conn.execute(text("SELECT id FROM users WHERE username = :u"), {"u": username}).fetchone()
            if user_res:
                user_id = user_res[0]
                conn.execute(
                    text("DELETE FROM favorites WHERE user_id = :uid AND article_id = :aid"),
                    {"uid": user_id, "aid": article_id}
                )
                conn.commit()
                return {"status": "success", "message": "Удалено из избранного"}
            return {"status": "error", "message": "Пользователь не найден"}
        except Exception as e:
            return {"status": "error", "message": str(e)}
