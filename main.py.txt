from fastapi import FastAPI
from sqlalchemy import create_engine, text

app = FastAPI()

DATABASE_URL = "postgresql://postgres:mysecretpassword@localhost:5432/history_db"
engine = create_engine(DATABASE_URL)

@app.get("/")
def read_root():
    return {"message": "Сервер работает! База данных подключена."}

@app.get("/articles")
def get_articles(topic: str = None):
    with engine.connect() as conn:
        if topic:
            query = text("SELECT id, title, content, topic FROM articles WHERE topic = :t")
            result = conn.execute(query, {"t": topic})
        else:
            query = text("SELECT id, title, content, topic FROM articles")
            result = conn.execute(query)
            
        articles = []
        for row in result:
            articles.append({
                "id": row[0],
                "title": row[1],
                "content": row[2],
                "topic": row[3]
            })
        return {"articles": articles}
